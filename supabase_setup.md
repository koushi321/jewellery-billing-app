# Supabase Database Setup Guide

Follow these steps to set up your Supabase database.

### Step 1: Create a Supabase Project
1. Go to [supabase.com](https://supabase.com) and sign up for a free account.
2. Click **New Project** and name it (e.g., `Toolshop`).
3. Set a secure **Database Password** (write this down, you may need it later).
4. Select a region close to you and click **Create new project**. It will take a minute or two to initialize.

### Step 2: Create Tables and Functions
Once the project is ready:
1. In the left sidebar of Supabase, click on the **SQL Editor** icon (looks like `>_`).
2. Click **New Query**.
3. Copy and paste the entire SQL block below into the editor.
4. Click **Run** (or press Ctrl + Enter / Cmd + Enter).

```sql
-- 1. Create Stock Table
CREATE TABLE IF NOT EXISTS stock (
  id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  name text NOT NULL,
  category text NOT NULL,
  box_name text,
  total_weight_g double precision NOT NULL,
  box_weight_g double precision NOT NULL,
  net_weight_g double precision NOT NULL,
  reorder_weight_g double precision DEFAULT 10,
  created_at timestamp with time zone DEFAULT timezone('utc'::text, now()) NOT NULL,
  updated_at timestamp with time zone DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- 2. Create Bills Table
CREATE TABLE IF NOT EXISTS bills (
  id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  bill_number text UNIQUE NOT NULL,
  customer_name text,
  customer_phone text,
  items_json jsonb NOT NULL,
  subtotal double precision NOT NULL,
  discount double precision DEFAULT 0,
  grand_total double precision NOT NULL,
  gold_rate double precision,
  silver_rate double precision,
  bill_date date NOT NULL,
  created_at timestamp with time zone DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- 3. Create Sales Table
CREATE TABLE IF NOT EXISTS sales (
  id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  bill_id bigint REFERENCES bills(id) ON DELETE CASCADE,
  item_id bigint REFERENCES stock(id) ON DELETE SET NULL,
  item_name text NOT NULL,
  category text NOT NULL,
  weight_sold_g double precision NOT NULL,
  rate_per_gram double precision NOT NULL,
  making_charges double precision DEFAULT 0,
  total_price double precision NOT NULL,
  sale_date date NOT NULL,
  created_at timestamp with time zone DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- 4. Create Gold Loans Table
CREATE TABLE IF NOT EXISTS gold_loans (
  id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  person_id text UNIQUE NOT NULL,
  person_name text NOT NULL,
  phone text,
  address text,
  gold_weight_g double precision NOT NULL,
  net_weight_g double precision DEFAULT 0,
  gold_purity text NOT NULL,
  gold_value double precision DEFAULT 0,
  loan_amount double precision NOT NULL,
  interest_rate double precision NOT NULL,
  loan_date date NOT NULL,
  due_date date,
  status text DEFAULT 'active',
  closed_date date,
  notes text,
  created_at timestamp with time zone DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- 5. Create Loan Payments Table
CREATE TABLE IF NOT EXISTS loan_payments (
  id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  loan_id bigint REFERENCES gold_loans(id) ON DELETE CASCADE,
  amount_paid double precision NOT NULL,
  payment_date date NOT NULL,
  notes text,
  created_at timestamp with time zone DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- 6. Create Weight Checks Table
CREATE TABLE IF NOT EXISTS weight_checks (
  id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  check_date date NOT NULL,
  item_id bigint REFERENCES stock(id) ON DELETE SET NULL,
  item_name text NOT NULL,
  expected_weight double precision NOT NULL,
  actual_weight double precision NOT NULL,
  difference double precision NOT NULL,
  status text NOT NULL,
  notes text,
  created_at timestamp with time zone DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- 7. Transactional Bill Function (RPC)
CREATE OR REPLACE FUNCTION create_bill_transaction(
  p_customer_name text,
  p_customer_phone text,
  p_items jsonb,
  p_old_items jsonb,
  p_subtotal double precision,
  p_discount double precision,
  p_grand_total double precision,
  p_bill_date date
) RETURNS jsonb AS $$
DECLARE
  v_bill_id bigint;
  v_bill_number text;
  v_year text;
  v_count integer;
  v_item jsonb;
  v_item_id bigint;
  v_actual_weight double precision;
BEGIN
  -- Generate bill number (BILL-YYYY-NNN)
  v_year := to_char(p_bill_date, 'YYYY');
  SELECT COUNT(*) + 1 INTO v_count FROM bills;
  v_bill_number := 'BILL-' || v_year || '-' || lpad(v_count::text, 3, '0');

  -- Insert the bill
  INSERT INTO bills (
    bill_number, customer_name, customer_phone,
    items_json, subtotal, discount, grand_total, bill_date
  ) VALUES (
    v_bill_number, p_customer_name, p_customer_phone,
    jsonb_build_object('items', p_items, 'old_items', p_old_items),
    p_subtotal, p_discount, p_grand_total, p_bill_date
  ) RETURNING id INTO v_bill_id;

  -- Loop through purchased items to save sales and update stock
  FOR v_item IN SELECT * FROM jsonb_array_elements(p_items)
  LOOP
    v_item_id := (v_item->>'item_id')::bigint;
    v_actual_weight := (v_item->>'actual_weight')::double precision;

    INSERT INTO sales (
      bill_id, item_id, item_name, category,
      weight_sold_g, rate_per_gram, making_charges,
      total_price, sale_date
    ) VALUES (
      v_bill_id,
      v_item_id,
      v_item->>'item_name',
      v_item->>'category',
      v_actual_weight,
      (v_item->>'rate')::double precision,
      coalesce((v_item->>'making_charges')::double precision, 0),
      (v_item->>'total_price')::double precision,
      p_bill_date
    );

    UPDATE stock
    SET
      total_weight_g = total_weight_g - v_actual_weight,
      net_weight_g = net_weight_g - v_actual_weight,
      updated_at = now()
    WHERE id = v_item_id;
  END LOOP;

  RETURN (SELECT row_to_json(b)::jsonb FROM bills b WHERE b.id = v_bill_id);
END;
$$ LANGUAGE plpgsql;
```

### Step 3: Enable Email / Password Sign In
1. In the Supabase Dashboard, click on the **Authentication** tab (key icon).
2. Go to **Providers** under the settings section.
3. Make sure **Email** is enabled, and **Confirm Email** is disabled (to make it simple to login).
4. Click on the **Users** tab inside Authentication.
5. Click **Add User** -> **Create User**.
6. Type the email (e.g. `admin@toolshop.com`) and password you want to use.
7. Click **Save**. This will be your admin login credentials for your jewellery shop!
